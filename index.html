// 渲染本月「每日單日損益」曲線圖 (Daily PnL Trend) function renderDailyCumulativeChart(monthlyRecords) { const ctx = document.getElementById('dailyCumulativeChart').getContext('2d'); if (dailyCumulativeChartInstance) { dailyCumulativeChartInstance.destroy(); } if (monthlyRecords.length === 0) { dailyCumulativeChartInstance = new Chart(ctx, { type: 'line', data: { labels: ['無資料'], datasets: [{ data: [0] }] }, options: { responsive: true, maintainAspectRatio: false } }); return; } // 按日期正序 (舊到新) 排序 const chronological = [...monthlyRecords].sort((a, b) => new Date(a.date) - new Date(b.date)); // 按日期加總「當天」的淨損益 (實現損益 - 手續費) const dailyNetMap = {}; chronological.forEach(r => { const day = r.date; const net = r.pnl - r.fee; dailyNetMap[day] = (dailyNetMap[day] || 0) + net; }); const sortedDates = Object.keys(dailyNetMap).sort(); const dailyData = sortedDates.map(date => dailyNetMap[date]); // 設定基準線（0元線）與顏色動態處理 dailyCumulativeChartInstance = new Chart(ctx, { type: 'line', data: { labels: sortedDates, datasets: [{ label: '單日淨損益 (NTD)', data: dailyData, borderColor: '#38bdf8', // 使用天藍色作為主線條 backgroundColor: 'rgba(56, 189, 248, 0.1)', borderWidth: 2, fill: true, tension: 0.2, // 微幅平滑曲線 pointRadius: 5, // 設定數據點顏色：當日獲利為綠色，虧損為紅色 pointBackgroundColor: dailyData.map(val => val >= 0 ? '#10b981' : '#f43f5e'), pointBorderColor: dailyData.map(val => val >= 0 ? '#10b981' : '#f43f5e') }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { labels: { color: '#cbd5e1' } }, tooltip: { callbacks: { label: function(context) { const val = context.parsed.y; const prefix = val > 0 ? '+' : ''; return ` 單日損益: ${prefix}$ ${val.toLocaleString()}`; } } } }, scales: { x: { ticks: { color: '#94a3b8' }, grid: { color: '#334155' } }, y: { ticks: { color: '#94a3b8', callback: function(value) { return (value > 0 ? '+' : '') + value.toLocaleString(); } }, grid: { // 將 0 軸顯示為明顯的白線，方便對比盈虧 color: (context) => context.tick.value === 0 ? '#94a3b8' : '#334155', lineWidth: (context) => context.tick.value === 0 ? 1.5 : 1 } } } } }); }
